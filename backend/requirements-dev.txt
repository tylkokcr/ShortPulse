# What it takes to check the backend, but not to run it.
#
# Split out of requirements.txt because the Dockerfile installs that file
# into the image that serves production. A test runner, a linter and a type
# checker have no business in a running container: they are weight, and
# they are attack surface for no return.
#
# One command for a dev environment — this pulls the runtime in too:
#
#     pip install -r requirements-dev.txt
#
-r requirements.txt

ruff>=0.5

# Config and rationale in pyproject.toml. asyncpg ships no annotations of
# its own, and the stubs are what let the checker see the database layer —
# where an argument in the wrong position is otherwise a runtime discovery.
mypy>=1.11
asyncpg-stubs>=0.29

# test_subtitle_engine opens the bundled fonts to check that the coverage
